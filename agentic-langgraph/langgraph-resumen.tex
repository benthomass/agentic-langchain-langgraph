\begin{concept}[title=¿Qué es LangGraph?]
Librería para construir agentes como \textbf{grafos de estado}: nodos = pasos computacionales,
aristas = control de flujo, estado = memoria compartida que se actualiza en cada paso.
\\[0.4em]
LangChain te da piezas (LLM, tools, prompts, mensajes).
LangGraph te da el \textbf{motor de orquestación} (bucles, bifurcaciones, persistencia).
\end{concept}

Desde \texttt{agentic-langgraph/}:
\texttt{uv sync} $\cdot$
\texttt{uv run python 01-react-agent/main.py}

\section{Modelo mental: State $\to$ Nodes $\to$ Edges $\to$ Graph}
\label{sec:mental}
% =============================================================================

\begin{mental}[title=Una frase]
Un grafo LangGraph es una máquina que: (1)~guarda un \textbf{estado}, (2)~pasa ese estado por \textbf{nodos},
(3)~cada nodo \textbf{devuelve un parche} (update parcial), (4)~LangGraph \textbf{fusiona} el parche con el estado,
(5)~las \textbf{aristas} eligen el siguiente nodo (o \texttt{END}).
\end{mental}

\begin{center}
\begin{tikzpicture}[
  box/.style={draw, rounded corners, align=center, font=\small, inner sep=6pt},
  arr/.style={-{Latex}, thick}
]
\node[box, fill=blue!8] (inv) {invoke\\\{\texttt{messages}: [...]\} };
\node[box, fill=green!8, right=1.1cm of inv] (st) {Estado\\actual};
\node[box, fill=orange!10, right=1.1cm of st] (nd) {Nodo\\lee + escribe};
\node[box, fill=violet!8, right=1.1cm of nd] (mg) {Reducer\\fusiona};
\node[box, fill=gray!15, right=1.1cm of mg] (nx) {Arista\\siguiente};
\draw[arr] (inv) -- (st);
\draw[arr] (st) -- (nd);
\draw[arr] (nd) -- (mg);
\draw[arr] (mg) -- (nx);
\draw[arr] (nx.south) -- ++(0,-0.55) -| node[pos=0.25, below, font=\scriptsize]{bucle} (st.south);
\end{tikzpicture}
\end{center}

\subsection{Piezas clave (vocabulario)}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \textbf{State}: TypedDict (o schema) con campos. En agentes de chat casi siempre hay \texttt{messages}.
  \item \textbf{Node}: función \texttt{state $\to$ update}. No mutas el estado a mano; \textbf{devuelves} un dict con lo que cambió.
  \item \textbf{Edge}: transición fija (\texttt{A $\to$ B}) o condicional (\texttt{A $\to$ f(state) $\to$ B|C|END}).
  \item \textbf{Reducer}: función que dice \emph{cómo} fusionar update + estado previo (clave para mensajes).
  \item \textbf{Compile}: congela el grafo en un \texttt{Runnable} invocable (\texttt{invoke}, \texttt{stream}, \ldots).
  \item \texttt{START}/\texttt{END}: sentinels de entrada/salida del grafo.
\end{itemize}

\begin{fromcode}[title=01-react-agent/main.py --- esqueleto]
\begin{lstlisting}
flow = StateGraph(MessagesState)
flow.add_node(AGENT_REASON, run_agent_reasoning)
flow.set_entry_point(AGENT_REASON)
flow.add_node(ACT, tool_node)
flow.add_conditional_edges(AGENT_REASON, should_continue, {END: END, ACT: ACT})
flow.add_edge(ACT, AGENT_REASON)
app = flow.compile()
\end{lstlisting}
\end{fromcode}

\section{Mensajes: el ``idioma'' del agente}
\label{sec:messages}
% =============================================================================

\begin{concept}[title=Por qué importan tanto]
En LangGraph (y LangChain) la conversación \textbf{no es un string}: es una lista tipada de mensajes.
Cada tipo tiene semántica distinta para el LLM y para el runtime de tools.
Si mezclas mal roles / tool\_calls / ToolMessage, el bucle se rompe.
\end{concept}

\subsection{Tipos que debes reconocer}
\begin{center}
\begin{tabular}{@{}llp{7.8cm}@{}}
\toprule
\textbf{Clase} & \textbf{Rol} & \textbf{Para qué} \\
\midrule
\texttt{HumanMessage} & user & Entrada del humano / prompt de usuario \\
\texttt{AIMessage} & assistant & Respuesta del LLM (texto y/o \texttt{tool\_calls}) \\
\texttt{SystemMessage} & system & Instrucciones de comportamiento \\
\texttt{ToolMessage} & tool & Resultado de ejecutar una tool (ligado a un \texttt{tool\_call\_id}) \\
\bottomrule
\end{tabular}
\end{center}

\begin{tip}[title=Equivalencias de formato]
También puedes pasar dicts \texttt{\{"role": "system", "content": "..."\}} al LLM.
Internamente LangChain los normaliza a \texttt{BaseMessage}.
En el estado del grafo, conviene trabajar con objetos mensaje (o dejar que \texttt{add\_messages} los convierta).
\end{tip}

\subsection{El ciclo tool-calling (lo crítico)}
Cuando el LLM decide usar tools, la secuencia en \texttt{state["messages"]} suele ser:

\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item \texttt{HumanMessage} --- pregunta del usuario.
  \item \texttt{AIMessage} con \texttt{tool\_calls=[...]} --- el modelo \emph{pide} tools (aún no las ejecuta).
  \item uno o más \texttt{ToolMessage} --- resultados reales de esas tools.
  \item nuevo \texttt{AIMessage} --- respuesta final (o más tool\_calls $\to$ se repite).
\end{enumerate}

\begin{warn}[title=El LLM no ejecuta tools]
\texttt{bind\_tools} solo enseña al modelo el esquema de las tools.
Quien ejecuta es tu código: en el curso, \texttt{ToolNode(tools)}.
Sin el nodo \texttt{act}, el \texttt{AIMessage} con \texttt{tool\_calls} se queda ``colgado''.
\end{warn}

\begin{fromcode}[title=Cómo arrancas el grafo (entrada = mensajes)]
\begin{lstlisting}
res = app.invoke({
    "messages": [
        HumanMessage(content="What is the weather in Tokyo? ...")
    ]
})
# salida: dict de estado; la respuesta final suele ser:
res["messages"][-1].content
\end{lstlisting}
\end{fromcode}

% =============================================================================
\section{Estado y reducers: cómo ``enviar'' mensajes}
\label{sec:state}
% =============================================================================

\subsection{Regla de oro}
Un nodo \textbf{no reemplaza} toda la lista de mensajes por defecto.
Devuelve un \textbf{update parcial}:

\begin{lstlisting}
return {"messages": [response]}   # "añade esto", no "borra todo y pon solo esto"
\end{lstlisting}

LangGraph fusiona ese update con el estado previo usando el \textbf{reducer} del campo.

\subsection{\texttt{add\_messages}: el reducer de chat}
\begin{concept}[title=\texttt{Annotated[list, add\_messages]}]
\texttt{add\_messages(left, right)} fusiona dos listas:
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item por defecto es \textbf{append-only} (acumula historial);
  \item si un mensaje nuevo tiene el \textbf{mismo \texttt{id}} que uno existente, lo \textbf{reemplaza}.
\end{itemize}
Por eso puedes devolver solo el mensaje nuevo y el historial completo sobrevive.
\end{concept}

Dos formas equivalentes en el curso:

\begin{center}
\begin{tabular}{@{}p{7.4cm}p{7.4cm}@{}}
\toprule
\textbf{Atajo (módulo 01)} & \textbf{Explícito (módulo 02)} \\
\midrule
\begin{lstlisting}
from langgraph.graph import MessagesState
# equivale a:
# messages: Annotated[list, add_messages]
StateGraph(MessagesState)
\end{lstlisting}
&
\begin{lstlisting}
class MessageGraph(TypedDict):
    messages: Annotated[
        list[BaseMessage],
        add_messages
    ]
\end{lstlisting}
\\
\bottomrule
\end{tabular}
\end{center}

\begin{warn}[title=Sin reducer = overwrite]
Si defines \texttt{messages: list[BaseMessage]} \emph{sin} \texttt{Annotated[..., add\_messages]},
el update suele \textbf{sobrescribir} el campo entero.
Pierdes el historial y rompes tool calling.
\end{warn}

\subsection{Qué ve / qué devuelve un nodo}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \textbf{Input}: el estado completo actual (p.\,ej.\ \texttt{state["messages"]} = historial).
  \item \textbf{Output}: dict parcial con claves del estado. Solo incluye lo que cambió.
  \item Otros campos (si los añades: \texttt{count}, \texttt{draft}, \texttt{critique},\ldots) pueden tener otros reducers
        (suma, last-value, append de listas, etc.).
\end{itemize}

\begin{fromcode}[title=01-react-agent/nodes.py --- nodo que añade un AIMessage]
\begin{lstlisting}
def run_agent_reasoning(state: MessagesState) -> MessagesState:
    response = llm.invoke(
        [{"role": "system", "content": SYSATEM_MESSAGE}, *state["messages"]]
    )
    return {"messages": [response]}  # se append-ea al historial
\end{lstlisting}
\end{fromcode}

\begin{mental}[title=Truco para no confundirte]
Piensa: ``el nodo propone \emph{deltas}; el reducer decide \emph{cómo se integran}.''
Para mensajes, el delta es ``estos mensajes nuevos''; el reducer es ``append (o replace by id)''.
\end{mental}

\subsection{Estado custom: más allá de solo mensajes}
Puedes enriquecer el TypedDict:

\begin{lstlisting}
class AgentState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages]
    iterations: int          # last-value (overwrite)
    draft: str               # último borrador
\end{lstlisting}

Posibilidades típicas: contadores de bucles, flags de ``ya está bien'', artefactos intermedios,
resultados estructurados, IDs de sesión, etc.

\section{Nodos, aristas y control de flujo}
\label{sec:flow}
% =============================================================================

\subsection{Nodos}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Función Python (o Runnable) asociada a un nombre string.
  \item Debe ser \textbf{puro respecto al contrato}: lee estado $\to$ calcula $\to$ return update.
  \item Puede encapsular LCEL (\texttt{prompt | llm}), tools, retrieval, APIs, etc.
\end{itemize}

\subsection{Aristas fijas vs condicionales}
\begin{center}
\begin{tabular}{@{}p{7.2cm}p{7.6cm}@{}}
\toprule
\textbf{Fija} & \textbf{Condicional} \\
\midrule
\texttt{add\_edge(ACT, AGENT\_REASON)}
siempre ACT $\to$ reason
&
\texttt{add\_conditional\_edges(src, router, mapping)}
el router mira el estado y elige clave del mapping
\\
\bottomrule
\end{tabular}
\end{center}

\begin{fromcode}[title=Router del ReAct]
\begin{lstlisting}
def should_continue(state: MessagesState) -> str:
    if not state["messages"][-1].tool_calls:
        return END      # respuesta final: no pide tools
    return ACT          # hay tool_calls: hay que ejecutarlas
\end{lstlisting}
\end{fromcode}

\begin{tip}[title=Patrón habitual]
Tras un nodo ``LLM'':
\begin{itemize}[leftmargin=*,itemsep=1pt]
  \item si hay \texttt{tool\_calls} $\to$ nodo tools;
  \item si no $\to$ \texttt{END} (o nodo de post-proceso).
\end{itemize}
Eso es exactamente el corazón del módulo 01.
\end{tip}

\subsection{\texttt{ToolNode}}
Prebuilt de LangGraph: lee el último \texttt{AIMessage}, ejecuta cada \texttt{tool\_call},
y emite los \texttt{ToolMessage} correspondientes (que el reducer append-ea).

\begin{lstlisting}
from langgraph.prebuilt import ToolNode
tool_node = ToolNode(tools)
\end{lstlisting}

\section{Módulo 01 --- Agente ReAct en grafo}
\label{sec:react}
% =============================================================================

\begin{concept}[title=Idea]
ReAct = \textbf{Reason} + \textbf{Act} en bucle.
El modelo razona (y opcionalmente pide tools); el grafo actúa; se vuelve a razonar con los resultados en el historial.
\end{concept}

\begin{center}
\begin{tikzpicture}[
  n/.style={draw, rounded corners, align=center, font=\small, inner sep=5pt, minimum width=2.4cm},
  arr/.style={-{Latex}, thick}
]
\node[n, fill=blue!10] (r) {agent\_reason\\LLM + tools};
\node[n, fill=orange!12, right=2.2cm of r] (a) {act\\ToolNode};
\node[n, fill=gray!18, below=1.0cm of r] (e) {END};
\draw[arr] (r) -- node[above, font=\scriptsize]{tool\_calls?} (a);
\draw[arr] (r) -- node[left, font=\scriptsize]{no} (e);
\draw[arr] (a) -- ++(0,-0.7) -| node[pos=0.25, below, font=\scriptsize]{siempre} (r);
\end{tikzpicture}
\end{center}

\subsection{Piezas en el repo}
\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \texttt{react.py}: tools (\texttt{TavilySearch}, \texttt{triple}) + \texttt{llm.bind\_tools(tools)}.
  \item \texttt{nodes.py}: nodo reason + \texttt{ToolNode}.
  \item \texttt{main.py}: grafo, router, \texttt{invoke} con \texttt{HumanMessage}.
\end{itemize}

\subsection{Qué ``viaja'' en cada paso (ejemplo mental)}
\begin{enumerate}[leftmargin=*,itemsep=2pt]
  \item \textbf{invoke}: \texttt{[Human(...)]}
  \item \textbf{reason}: LLM ve system + historial $\to$ append \texttt{AIMessage(tool\_calls=[search,...])}
  \item router $\to$ \texttt{act}
  \item \textbf{act}: ToolNode $\to$ append \texttt{ToolMessage(content=...)}
  \item vuelta a \textbf{reason}: ahora el historial incluye resultados $\to$ quizá más tools o respuesta final
  \item si \texttt{tool\_calls} vacío $\to$ \texttt{END}
\end{enumerate}

\begin{tip}[title=Por qué el historial es obligatorio]
Sin append de \texttt{ToolMessage}, el LLM no ``ve'' el resultado de la tool en el siguiente turno.
El estado \emph{es} la memoria de corto plazo del agente.
\end{tip}

\section{Módulo 02 --- Reflection agent}
\label{sec:reflect}
% =============================================================================

\begin{concept}[title=Idea]
Dos roles en bucle: un \textbf{generador} produce un draft; un \textbf{reflector} critica;
el generador revisa con esa crítica en el historial.
No hace falta tools externas: el ``acto'' es criticar/mejorar texto.
\end{concept}

\subsection{Chains (LCEL) como cerebro de nodos}
\begin{fromcode}[title=02-reflection-agent/chains.py]
\begin{lstlisting}
generation_prompt = ChatPromptTemplate.from_messages([
    ("system", "... escribe el mejor tweet ..."),
    MessagesPlaceholder(variable_name="messages"),
])
generate_chain = generation_prompt | llm
# reflection_chain análogo, con prompt de crítico
\end{lstlisting}
\end{fromcode}

\texttt{MessagesPlaceholder} inyecta \textbf{toda} la lista \texttt{state["messages"]} en el prompt.
Así el generador ve request + críticas previas; el reflector ve el draft a evaluar.

\subsection{Nodo generate (patrón)}
\begin{lstlisting}
def generate_node(state):
    response = generate_chain.invoke({"messages": state["messages"]})
    # o generate_chain.invoke(state["messages"]) según cómo esté el prompt
    return {"messages": [response]}
\end{lstlisting}

\subsection{Flujo típico Reflection (cuando completes el grafo)}
\begin{center}
\begin{tikzpicture}[
  n/.style={draw, rounded corners, align=center, font=\small, inner sep=5pt, minimum width=2.2cm},
  arr/.style={-{Latex}, thick}
]
\node[n, fill=blue!10] (g) {generate};
\node[n, fill=violet!12, right=2.0cm of g] (r) {reflect};
\node[n, fill=gray!18, below=1.0cm of g] (e) {END};
\draw[arr] (g) -- (r);
\draw[arr] (r) -- ++(0,0.75) -| node[pos=0.2, above, font=\scriptsize]{sigue} (g);
\draw[arr] (g) -- node[left, font=\scriptsize]{¿listo?} (e);
\end{tikzpicture}
\end{center}

\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item Tras \texttt{generate}: ¿parar o reflejar? (contador de iteraciones, o heurística).
  \item Tras \texttt{reflect}: normalmente vuelves a \texttt{generate} (la crítica entra como mensaje más).
  \item Truco frecuente: en reflect, devolver la crítica como \texttt{HumanMessage} (o role user)
        para que el generador la trate como feedback externo.
\end{itemize}

\begin{warn}[title=Estado del código del módulo 02]
El grafo aún está incompleto en el repo (\texttt{main.py} define estado + \texttt{generate\_node}, pero no monta edges/compile).
Hay un typo de import: \texttt{langgrapg} $\to$ \texttt{langgraph}.
El resumen describe el \textbf{patrón}, no asume que el ejemplo compile tal cual.
\end{warn}

\section{Posibilidades: qué puedes construir con esto}
\label{sec:possibilities}
% =============================================================================

Con solo State + Nodes + Conditional edges ya cubres una familia grande de agentes:

\begin{itemize}[leftmargin=*,itemsep=3pt]
  \item \textbf{Tool agent (ReAct)}: LLM $\leftrightarrow$ tools hasta respuesta final.
  \item \textbf{Reflection / critique}: generate $\leftrightarrow$ critique con historial compartido.
  \item \textbf{Multi-agente}: nodos distintos (researcher, writer, reviewer) + handoff por aristas.
  \item \textbf{RAG en un nodo}: retrieval dentro de un nodo; el resto del grafo orquesta.
  \item \textbf{Human-in-the-loop}: arista a un nodo que espera input humano (con checkpointer).
  \item \textbf{Routing}: clasificar la query y mandarla a subgrafos distintos.
  \item \textbf{Map-reduce}: fan-out paralelo a varios nodos y join (LangGraph lo soporta).
\end{itemize}

\begin{tip}[title=LangGraph vs agent ``caja negra'']
\texttt{create\_agent} (LangChain) oculta el bucle.
Con LangGraph \textbf{ves y diseñas} el bucle: cuándo tools, cuándo parar, cuándo criticar, qué guardar en estado.
\end{tip}

\section{Errores mentales frecuentes}
\label{sec:pitfalls}
% =============================================================================

\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item \textbf{``return messages = historial completo''}: innecesario (y peligroso) si usas \texttt{add\_messages}; devuelve solo lo nuevo.
  \item \textbf{Olvidar el reducer}: sin \texttt{add\_messages}, pisas el historial.
  \item \textbf{Confundir AIMessage.tool\_calls con ejecución}: pedir $\neq$ ejecutar.
  \item \textbf{No volver al LLM tras ToolMessage}: el grafo debe regresar al nodo reason.
  \item \textbf{Meter system prompt solo una vez en el estado}: a veces es mejor inyectarlo en el invoke del nodo (como en 01) y no ensuciar el historial persistente.
  \item \textbf{Reflection sin criterio de parada}: bucle infinito generate$\leftrightarrow$reflect; usa contador o ``critique vacía / OK''.
\end{enumerate}
